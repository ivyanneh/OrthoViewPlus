import fs from 'node:fs';
import path from 'node:path';
import { AppError } from '../utils/http.js';
import { writeResultFile } from './aiClient.js';

const SELECT_JOB = `
  SELECT j.id, j.capability, j.status, j.requested_by AS requestedBy, j.patient_id AS patientId,
         j.source_study_id AS sourceStudyId, j.orthanc_instance_id AS orthancInstanceId,
         j.input_json AS inputJson, j.error, j.created_at AS createdAt, j.started_at AS startedAt,
         j.completed_at AS completedAt
  FROM ai_jobs j
`;

const SELECT_RESULT = `
  SELECT id, job_id AS jobId, capability, source_study_id AS sourceStudyId, result_kind AS resultKind,
         file_path AS filePath, overlay_path AS overlayPath, metadata_json AS metadataJson,
         label, review_status AS reviewStatus, created_at AS createdAt
  FROM ai_results
`;

export function createAiJobService(db, { aiClient, orthanc, resultsDirectory, audit }) {
  function getJob(id) {
    const job = db.prepare(`${SELECT_JOB} WHERE j.id = ?`).get(Number(id));
    if (!job) {
      return null;
    }
    return {
      ...job,
      input: job.inputJson ? JSON.parse(job.inputJson) : null,
      results: db.prepare(`${SELECT_RESULT} WHERE job_id = ?`).all(job.id).map(publicResult),
    };
  }

  function publicResult(row) {
    return {
      ...row,
      metadata: row.metadataJson ? JSON.parse(row.metadataJson) : null,
      fileUrl: row.filePath ? `/api/ai/results/${row.id}/file` : null,
      overlayUrl: row.overlayPath ? `/api/ai/results/${row.id}/overlay` : null,
    };
  }

  async function collectFiles({ instanceId, priorInstanceId, seriesId }) {
    const files = [];
    if (instanceId) {
      const buffer = await orthanc.getInstanceFile(instanceId);
      files.push({ field: 'image', buffer, filename: `${instanceId}.dcm`, contentType: 'application/dicom' });
    }
    if (priorInstanceId) {
      const buffer = await orthanc.getInstanceFile(priorInstanceId);
      files.push({ field: 'prior_image', buffer, filename: `${priorInstanceId}.dcm`, contentType: 'application/dicom' });
    }
    if (seriesId) {
      const series = await orthanc.getSeries(seriesId);
      const instanceIds = Array.isArray(series.Instances) ? series.Instances.slice(0, 80) : [];
      for (const id of instanceIds) {
        const buffer = await orthanc.getInstanceFile(id);
        files.push({ field: 'volume', buffer, filename: `${id}.dcm`, contentType: 'application/dicom' });
      }
    }
    return files;
  }

  async function runJob(jobId) {
    const jobRow = db.prepare('SELECT * FROM ai_jobs WHERE id = ?').get(jobId);
    if (!jobRow) {
      return;
    }
    db.prepare(`UPDATE ai_jobs SET status = 'running', started_at = datetime('now') WHERE id = ?`).run(jobId);
    const input = jobRow.input_json ? JSON.parse(jobRow.input_json) : {};
    try {
      const files = jobRow.capability === 'synthetic' ? [] : await collectFiles(input);
      const output = await aiClient.process({
        capability: jobRow.capability,
        files,
        fields: {
          exam_type: input.examType || '',
          patient_id: input.patientId || '',
          prior_patient_id: input.priorPatientId || '',
          body_part: input.bodyPart || '',
          notes: input.notes || '',
        },
      });

      const saved = [];
      if (output.image_base64) {
        saved.push(
          writeResultFile(resultsDirectory, `job-${jobId}-primary.png`, output.image_base64),
        );
      }
      if (output.overlay_base64) {
        saved.push(
          writeResultFile(resultsDirectory, `job-${jobId}-overlay.png`, output.overlay_base64),
        );
      }
      if (output.mesh) {
        const meshPath = path.join(resultsDirectory, `job-${jobId}-mesh.json`);
        fs.mkdirSync(resultsDirectory, { recursive: true });
        fs.writeFileSync(meshPath, JSON.stringify(output.mesh));
        saved.unshift(meshPath);
      }

      db.prepare(
        `INSERT INTO ai_results (job_id, capability, source_study_id, result_kind, file_path, overlay_path, metadata_json, label)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
      ).run(
        jobId,
        jobRow.capability,
        jobRow.source_study_id,
        output.result_kind || jobRow.capability,
        saved[0] || null,
        output.overlay_base64 ? path.join(resultsDirectory, `job-${jobId}-overlay.png`) : null,
        JSON.stringify({
          method: output.method,
          licence: output.licence,
          warnings: output.warnings || [],
          metrics: output.metrics || {},
          experimental: Boolean(output.experimental),
          synthetic: Boolean(output.synthetic),
          meshIncluded: Boolean(output.mesh),
        }),
        output.displayLabel || output.label,
      );

      db.prepare(`UPDATE ai_jobs SET status = 'completed', completed_at = datetime('now'), error = NULL WHERE id = ?`).run(jobId);
      audit?.log({
        userId: jobRow.requested_by,
        action: 'ai_job_completed',
        target: `ai_job:${jobId}`,
        description: `AI job ${jobId} (${jobRow.capability}) completed.`,
      });
    } catch (error) {
      db.prepare(`UPDATE ai_jobs SET status = 'failed', completed_at = datetime('now'), error = ? WHERE id = ?`).run(
        error.message || 'AI processing failed.',
        jobId,
      );
      audit?.log({
        userId: jobRow.requested_by,
        action: 'ai_job_failed',
        target: `ai_job:${jobId}`,
        description: `AI job ${jobId} failed: ${error.message}`,
      });
    }
  }

  return {
    getJob,

    list({ requestedBy, limit = 50 } = {}) {
      const rows = db
        .prepare(`${SELECT_JOB} WHERE j.requested_by = @requestedBy ORDER BY j.created_at DESC LIMIT @limit`)
        .all({ requestedBy: Number(requestedBy), limit: Math.min(Number(limit) || 50, 200) });
      return rows.map((job) => ({
        ...job,
        input: job.inputJson ? JSON.parse(job.inputJson) : null,
      }));
    },

    getResult(id) {
      const row = db.prepare(`${SELECT_RESULT} WHERE id = ?`).get(Number(id));
      return row ? publicResult(row) : null;
    },

    markReviewed(id, userId) {
      const result = this.getResult(id);
      if (!result) {
        throw new AppError('AI result not found.', 404);
      }
      db.prepare(`UPDATE ai_results SET review_status = 'reviewed' WHERE id = ?`).run(Number(id));
      audit?.log({
        userId,
        action: 'ai_result_reviewed',
        target: `ai_result:${id}`,
        description: `Radiologist reviewed AI result ${id}.`,
      });
      return this.getResult(id);
    },

    enqueue({ capability, requestedBy, patientId, sourceStudyId, input, auditUser }) {
      if (!aiClient.capabilities[capability]) {
        throw new AppError('Unsupported AI capability.');
      }
      if (capability !== 'synthetic' && !input?.instanceId && !input?.seriesId) {
        throw new AppError('An Orthanc instance or series is required for this capability.');
      }
      const result = db
        .prepare(
          `INSERT INTO ai_jobs (capability, requested_by, patient_id, source_study_id, orthanc_instance_id, input_json)
           VALUES (?, ?, ?, ?, ?, ?)`,
        )
        .run(
          capability,
          requestedBy,
          patientId ?? null,
          sourceStudyId ?? null,
          input?.instanceId ?? null,
          JSON.stringify(input || {}),
        );
      audit?.log({
        userId: auditUser,
        action: 'ai_job_queued',
        target: `ai_job:${result.lastInsertRowid}`,
        description: `Queued AI capability "${capability}".`,
      });
      setImmediate(() => {
        runJob(result.lastInsertRowid).catch((error) => {
          console.error('AI job failed', error);
        });
      });
      return getJob(result.lastInsertRowid);
    },
  };
}

import fs from 'node:fs';
import path from 'node:path';
import axios from 'axios';
import FormData from 'form-data';
import { AppError } from '../utils/http.js';

const CAPABILITIES = {
  restore: { endpoint: '/api/ai/restore', label: 'AI-restored image — not a substitute for the original acquisition' },
  completeness: { endpoint: '/api/ai/completeness-check', label: 'Coverage assessment — assistive, requires review' },
  compare: { endpoint: '/api/ai/compare', label: 'Longitudinal comparison — aligned difference map' },
  reconstruct: { endpoint: '/api/ai/reconstruct', label: 'Estimated reconstruction — generated regions, not exact recovery' },
  segment3d: { endpoint: '/api/ai/segment-3d', label: '3D visualisation derived from CT volume — experimental where noted' },
  synthetic: { endpoint: '/api/ai/generate-synthetic', label: 'SYNTHETIC image — not a patient examination' },
  complementary: { endpoint: '/api/ai/generate-complementary-view', label: 'SYNTHETIC complementary view — estimated, not acquired' },
};

export function createAiClient({ baseUrl, timeoutMs = 120000 }) {
  const client = axios.create({
    baseURL: String(baseUrl || '').replace(/\/$/, ''),
    timeout: timeoutMs,
    validateStatus: (status) => status >= 200 && status < 500,
  });

  return {
    capabilities: CAPABILITIES,

    async health() {
      try {
        const response = await client.get('/health');
        return { available: response.status === 200, ...response.data };
      } catch {
        return { available: false, status: 'AI service unavailable' };
      }
    },

    async process({ capability, files, fields }) {
      const spec = CAPABILITIES[capability];
      if (!spec) {
        throw new AppError('Unsupported AI capability.');
      }
      const form = new FormData();
      for (const [key, value] of Object.entries(fields || {})) {
        if (value !== undefined && value !== null) {
          form.append(key, typeof value === 'string' ? value : JSON.stringify(value));
        }
      }
      for (const file of files || []) {
        form.append(file.field, file.buffer, { filename: file.filename, contentType: file.contentType || 'application/octet-stream' });
      }
      try {
        const response = await client.post(spec.endpoint, form, {
          headers: form.getHeaders(),
          maxBodyLength: Infinity,
          maxContentLength: Infinity,
        });
        if (response.status >= 400) {
          throw new AppError(response.data?.detail || response.data?.error || 'AI processing failed.', response.status);
        }
        return { ...response.data, displayLabel: spec.label };
      } catch (error) {
        if (error instanceof AppError) {
          throw error;
        }
        if (!error.response) {
          throw new AppError('The AI processing service is not available.', 503);
        }
        throw new AppError(error.response.data?.detail || error.response.data?.error || 'AI processing failed.', 502);
      }
    },
  };
}

export function writeResultFile(directory, filename, base64Data) {
  fs.mkdirSync(directory, { recursive: true });
  const filePath = path.join(directory, filename);
  fs.writeFileSync(filePath, Buffer.from(base64Data, 'base64'));
  return filePath;
}

import { AppError, cleanText } from '../utils/http.js';

const SELECT = `
  SELECT p.id, p.mrn, p.first_name AS firstName, p.last_name AS lastName,
         p.date_of_birth AS dateOfBirth, p.sex, p.notes, p.status,
         p.created_at AS createdAt, p.updated_at AS updatedAt,
         creator.username AS createdBy
  FROM patients p
  LEFT JOIN users creator ON creator.id = p.created_by
`;

export function createPatientService(db) {
  const getById = db.prepare(`${SELECT} WHERE p.id = ?`);

  function get(id) {
    return getById.get(Number(id)) ?? null;
  }

  function requirePatient(id) {
    const patient = get(id);
    if (!patient) {
      throw new AppError('Patient not found.', 404);
    }
    return patient;
  }

  return {
    get,
    requirePatient,

    search({ query = '', status = 'active', limit = 50 } = {}) {
      const clauses = [];
      const params = { limit: Math.min(Math.max(Number(limit) || 50, 1), 200) };
      if (status && status !== 'all') {
        clauses.push('p.status = @status');
        params.status = status;
      }
      if (query) {
        clauses.push(`(
          LOWER(p.mrn) LIKE @query OR
          LOWER(p.first_name) LIKE @query OR
          LOWER(p.last_name) LIKE @query OR
          LOWER(p.first_name || ' ' || p.last_name) LIKE @query
        )`);
        params.query = `%${String(query).toLowerCase()}%`;
      }
      const where = clauses.length ? `WHERE ${clauses.join(' AND ')}` : '';
      return db
        .prepare(`${SELECT} ${where} ORDER BY p.last_name COLLATE NOCASE, p.first_name COLLATE NOCASE LIMIT @limit`)
        .all(params);
    },

    create({ mrn, firstName, lastName, dateOfBirth, sex, notes, createdBy }) {
      const record = {
        mrn: cleanText(mrn, 40),
        firstName: cleanText(firstName, 80),
        lastName: cleanText(lastName, 80),
        dateOfBirth: cleanText(dateOfBirth, 10) ?? null,
        sex: cleanText(sex, 20) ?? null,
        notes: cleanText(notes, 500) ?? null,
      };
      if (!record.mrn || !record.firstName || !record.lastName) {
        throw new AppError('MRN, first name and last name are required.');
      }
      if (db.prepare('SELECT id FROM patients WHERE mrn = ?').get(record.mrn)) {
        throw new AppError('A patient with that MRN already exists.', 409);
      }
      const result = db
        .prepare(
          `INSERT INTO patients (mrn, first_name, last_name, date_of_birth, sex, notes, created_by, updated_at)
           VALUES (?, ?, ?, ?, ?, ?, ?, datetime('now'))`,
        )
        .run(record.mrn, record.firstName, record.lastName, record.dateOfBirth, record.sex, record.notes, createdBy ?? null);
      return get(result.lastInsertRowid);
    },

    update(id, changes) {
      const patient = requirePatient(id);
      if (patient.status === 'archived') {
        throw new AppError('Archived patient records cannot be edited.', 409);
      }
      const next = {
        firstName: changes.firstName !== undefined ? cleanText(changes.firstName, 80) : patient.firstName,
        lastName: changes.lastName !== undefined ? cleanText(changes.lastName, 80) : patient.lastName,
        dateOfBirth: changes.dateOfBirth !== undefined ? cleanText(changes.dateOfBirth, 10) : patient.dateOfBirth,
        sex: changes.sex !== undefined ? cleanText(changes.sex, 20) : patient.sex,
        notes: changes.notes !== undefined ? cleanText(changes.notes, 500) : patient.notes,
      };
      if (!next.firstName || !next.lastName) {
        throw new AppError('First name and last name are required.');
      }
      db.prepare(
        `UPDATE patients SET first_name = ?, last_name = ?, date_of_birth = ?, sex = ?, notes = ?, updated_at = datetime('now') WHERE id = ?`,
      ).run(next.firstName, next.lastName, next.dateOfBirth, next.sex, next.notes, patient.id);
      return get(patient.id);
    },

    archive(id) {
      const patient = requirePatient(id);
      db.prepare(`UPDATE patients SET status = 'archived', updated_at = datetime('now') WHERE id = ?`).run(patient.id);
      return get(patient.id);
    },

    count() {
      return db.prepare(`SELECT COUNT(*) AS total FROM patients WHERE status = 'active'`).get().total;
    },
  };
}

import { AppError, cleanText } from '../utils/http.js';

const SELECT = `
  SELECT r.id, r.patient_id AS patientId, r.appointment_id AS appointmentId,
         r.requested_by AS requestedBy, r.assigned_radiologist_id AS assignedRadiologistId,
         r.examination, r.modality, r.priority, r.clinical_indication AS clinicalIndication,
         r.status, r.created_at AS createdAt, r.updated_at AS updatedAt,
         p.mrn, p.first_name AS firstName, p.last_name AS lastName,
         requester.username AS requestedByUsername,
         COALESCE(radiologist.full_name, radiologist.username) AS assignedRadiologist
  FROM imaging_requests r
  JOIN patients p ON p.id = r.patient_id
  LEFT JOIN users requester ON requester.id = r.requested_by
  LEFT JOIN users radiologist ON radiologist.id = r.assigned_radiologist_id
`;

export function createRequestService(db, { notifications } = {}) {
  const getById = db.prepare(`${SELECT} WHERE r.id = ?`);

  function get(id) {
    return getById.get(Number(id)) ?? null;
  }

  function list({ status, patientId, assignedRadiologistId, requestedBy, limit = 200 } = {}) {
    const clauses = [];
    const params = { limit: Math.min(Math.max(Number(limit) || 200, 1), 500) };
    if (status && status !== 'all') {
      clauses.push('r.status = @status');
      params.status = status;
    }
    if (patientId) {
      clauses.push('r.patient_id = @patientId');
      params.patientId = Number(patientId);
    }
    if (assignedRadiologistId) {
      clauses.push('(r.assigned_radiologist_id = @assignedRadiologistId OR r.assigned_radiologist_id IS NULL)');
      params.assignedRadiologistId = Number(assignedRadiologistId);
    }
    if (requestedBy) {
      clauses.push('r.requested_by = @requestedBy');
      params.requestedBy = Number(requestedBy);
    }
    const where = clauses.length ? `WHERE ${clauses.join(' AND ')}` : '';
    return db.prepare(`${SELECT} ${where} ORDER BY r.created_at DESC, r.id DESC LIMIT @limit`).all(params);
  }

  return {
    get,
    list,

    create({ patientId, requestedBy, appointmentId, examination, modality, priority = 'routine', clinicalIndication, assignedRadiologistId }) {
      const patient = db.prepare('SELECT id, first_name, last_name FROM patients WHERE id = ? AND status = ?').get(Number(patientId), 'active');
      if (!patient) {
        throw new AppError('Patient not found or archived.', 404);
      }
      const exam = cleanText(examination, 120);
      const mod = String(modality || '').trim().toUpperCase();
      if (!exam || !mod) {
        throw new AppError('Examination and modality are required.');
      }
      if (!['routine', 'urgent'].includes(priority)) {
        throw new AppError('Priority must be routine or urgent.');
      }
      const result = db
        .prepare(
          `INSERT INTO imaging_requests (
             patient_id, appointment_id, requested_by, assigned_radiologist_id,
             examination, modality, priority, clinical_indication, updated_at
           ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, datetime('now'))`,
        )
        .run(
          patient.id,
          appointmentId ? Number(appointmentId) : null,
          requestedBy,
          assignedRadiologistId ? Number(assignedRadiologistId) : null,
          exam,
          mod,
          priority,
          cleanText(clinicalIndication, 500) ?? null,
        );

      notifications?.notifyRole('radiologist', {
        type: 'imaging_request',
        title: 'New imaging request',
        message: `${patient.first_name} ${patient.last_name}: ${exam} (${mod}).`,
        resourceType: 'imaging_request',
        resourceId: result.lastInsertRowid,
      });

      return get(result.lastInsertRowid);
    },

    setStatus(id, status) {
      const request = get(Number(id));
      if (!request) {
        throw new AppError('Imaging request not found.', 404);
      }
      const allowed = ['submitted', 'scheduled', 'in_progress', 'completed', 'cancelled'];
      if (!allowed.includes(status)) {
        throw new AppError('Invalid imaging request status.');
      }
      db.prepare(`UPDATE imaging_requests SET status = ?, updated_at = datetime('now') WHERE id = ?`).run(status, request.id);
      return get(request.id);
    },

    assign(id, radiologistId) {
      const request = get(Number(id));
      if (!request) {
        throw new AppError('Imaging request not found.', 404);
      }
      db.prepare(
        `UPDATE imaging_requests SET assigned_radiologist_id = ?, status = CASE WHEN status = 'submitted' THEN 'in_progress' ELSE status END, updated_at = datetime('now') WHERE id = ?`,
      ).run(Number(radiologistId), request.id);
      return get(request.id);
    },
  };
}

import { AppError, cleanText } from '../utils/http.js';

const SELECT = `
  SELECT r.id, r.imaging_study_id AS imagingStudyId, r.patient_id AS patientId,
         r.imaging_request_id AS imagingRequestId, r.radiologist_id AS radiologistId,
         r.findings, r.impression, r.status, r.created_at AS createdAt, r.updated_at AS updatedAt,
         r.approved_at AS approvedAt,
         p.mrn, p.first_name AS firstName, p.last_name AS lastName,
         COALESCE(rad.full_name, rad.username) AS radiologistName
  FROM radiology_reports r
  JOIN patients p ON p.id = r.patient_id
  LEFT JOIN users rad ON rad.id = r.radiologist_id
`;

export function createReportService(db, { notifications } = {}) {
  const getById = db.prepare(`${SELECT} WHERE r.id = ?`);

  function get(id) {
    return getById.get(Number(id)) ?? null;
  }

  function list({ patientId, status, radiologistId, limit = 100 } = {}) {
    const clauses = [];
    const params = { limit: Math.min(Math.max(Number(limit) || 100, 1), 300) };
    if (patientId) {
      clauses.push('r.patient_id = @patientId');
      params.patientId = Number(patientId);
    }
    if (status && status !== 'all') {
      clauses.push('r.status = @status');
      params.status = status;
    }
    if (radiologistId) {
      clauses.push('r.radiologist_id = @radiologistId');
      params.radiologistId = Number(radiologistId);
    }
    const where = clauses.length ? `WHERE ${clauses.join(' AND ')}` : '';
    return db.prepare(`${SELECT} ${where} ORDER BY r.updated_at DESC, r.id DESC LIMIT @limit`).all(params);
  }

  return {
    get,
    list,

    saveDraft({ id, patientId, imagingStudyId, imagingRequestId, radiologistId, findings, impression }) {
      const payload = {
        findings: cleanText(findings, 8000) ?? '',
        impression: cleanText(impression, 4000) ?? '',
      };
      if (id) {
        const existing = get(Number(id));
        if (!existing) {
          throw new AppError('Report not found.', 404);
        }
        if (existing.status === 'approved') {
          throw new AppError('Approved reports cannot be edited.', 409);
        }
        if (Number(existing.radiologistId) !== Number(radiologistId)) {
          throw new AppError('You can only edit your own draft reports.', 403);
        }
        db.prepare(
          `UPDATE radiology_reports SET findings = ?, impression = ?, imaging_study_id = ?, updated_at = datetime('now') WHERE id = ?`,
        ).run(payload.findings, payload.impression, imagingStudyId ? Number(imagingStudyId) : existing.imagingStudyId, existing.id);
        return get(existing.id);
      }

      const patient = db.prepare('SELECT id FROM patients WHERE id = ?').get(Number(patientId));
      if (!patient) {
        throw new AppError('Patient not found.', 404);
      }
      const result = db
        .prepare(
          `INSERT INTO radiology_reports (
             imaging_study_id, patient_id, imaging_request_id, radiologist_id, findings, impression, updated_at
           ) VALUES (?, ?, ?, ?, ?, ?, datetime('now'))`,
        )
        .run(
          imagingStudyId ? Number(imagingStudyId) : null,
          patient.id,
          imagingRequestId ? Number(imagingRequestId) : null,
          radiologistId,
          payload.findings,
          payload.impression,
        );
      return get(result.lastInsertRowid);
    },

    approve(id, radiologistId) {
      const report = get(Number(id));
      if (!report) {
        throw new AppError('Report not found.', 404);
      }
      if (report.status === 'approved') {
        throw new AppError('This report is already approved.', 409);
      }
      if (Number(report.radiologistId) !== Number(radiologistId)) {
        throw new AppError('Only the authoring radiologist can approve this report.', 403);
      }
      if (!report.findings && !report.impression) {
        throw new AppError('Add findings or an impression before approving.');
      }
      db.prepare(
        `UPDATE radiology_reports SET status = 'approved', approved_at = datetime('now'), updated_at = datetime('now') WHERE id = ?`,
      ).run(report.id);

      notifications?.notifyRole('practitioner', {
        type: 'report_ready',
        title: 'Radiology report available',
        message: `An approved report is available for ${report.firstName} ${report.lastName}.`,
        resourceType: 'report',
        resourceId: report.id,
      });

      return get(report.id);
    },
  };
}

const SELECT = `

         p.mrn, p.first_name AS firstName, p.last_name AS lastName,
         COALESCE(assignee.full_name, assignee.username) AS assignedTo,
         requester.username AS requestedByUsername
  FROM appointments a
  JOIN patients p ON p.id = a.patient_id
  LEFT JOIN users assignee ON assignee.id = a.assigned_user_id
  LEFT JOIN users requester ON requester.id = a.requested_by
`;

export function createAppointmentService(db, { notifications, alerts } = {}) {
  const getById = db.prepare(`${SELECT} WHERE a.id = ?`);

  function get(id) {
    return getById.get(Number(id)) ?? null;
  }

  function requireAppointment(id) {
    const appointment = get(id);
    if (!appointment) {
      throw new AppError('Appointment not found.', 404);
    }
    return appointment;
  }

  function validate(input) {
    const date = String(input.date || '').trim();
    const time = String(input.time || '').trim();
    if (!isValidDate(date)) {
      throw new AppError('Enter a valid appointment date.');
    }
    if (!/^\d{2}:\d{2}$/.test(time)) {
      throw new AppError('Enter a valid time in HH:MM format.');
    }
    const examination = cleanText(input.examination, 120);
    const modality = String(input.modality || '').trim().toUpperCase();
    if (!examination || !modality) {
      throw new AppError('Examination and modality are required.');
    }
    return {
      date,
      time,
      examination,
      modality,
      notes: cleanText(input.notes, 400) ?? null,
      assignedUserId: input.assignedUserId ? Number(input.assignedUserId) : null,
    };
  }

  function list({ status, dateFrom, dateTo, patientId, limit = 200 } = {}) {
    const clauses = [];
    const params = { limit: Math.min(Math.max(Number(limit) || 200, 1), 500) };
    if (status && status !== 'all') {
      clauses.push('a.status = @status');
      params.status = status;
    }
    if (dateFrom) {
      clauses.push('a.scheduled_date >= @dateFrom');
      params.dateFrom = dateFrom;
    }
    if (dateTo) {
      clauses.push('a.scheduled_date <= @dateTo');
      params.dateTo = dateTo;
    }
    if (patientId) {
      clauses.push('a.patient_id = @patientId');
      params.patientId = Number(patientId);
    }
    const where = clauses.length ? `WHERE ${clauses.join(' AND ')}` : '';
    return db
      .prepare(`${SELECT} ${where} ORDER BY a.scheduled_date, a.scheduled_time, a.id LIMIT @limit`)
      .all(params);
  }

  function counts() {
    const rows = db.prepare(`SELECT status, COUNT(*) AS total FROM appointments GROUP BY status`).all();
    const summary = { scheduled: 0, completed: 0, cancelled: 0, missed: 0 };
    for (const row of rows) {
      summary[row.status] = row.total;
    }
    return summary;
  }

  return {
    get,
    list,
    counts,

    create({ patientId, requestedBy, createdBy, ...input }) {
      const patient = db.prepare('SELECT id FROM patients WHERE id = ? AND status = ?').get(Number(patientId), 'active');
      if (!patient) {
        throw new AppError('Patient not found or archived.', 404);
      }
      const entry = validate(input);
      const result = db
        .prepare(
          `INSERT INTO appointments (
             patient_id, scheduled_date, scheduled_time, examination, modality,
             assigned_user_id, requested_by, notes, created_by, updated_at
           ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, datetime('now'))`,
        )
        .run(
          patient.id,
          entry.date,
          entry.time,
          entry.examination,
          entry.modality,
          entry.assignedUserId,
          requestedBy ?? null,
          entry.notes,
          createdBy ?? null,
        );
      return get(result.lastInsertRowid);
    },

    setStatus(id, status, actorId) {
      const appointment = requireAppointment(id);
      if (!APPOINTMENT_STATUSES.includes(status)) {
        throw new AppError('Invalid appointment status.');
      }
      if (appointment.status === 'cancelled' && status !== 'cancelled') {
        throw new AppError('Cancelled appointments cannot be reopened.', 409);
      }
      const extra =
        status === 'completed'
          ? `, completed_at = datetime('now')`
          : status === 'missed'
            ? `, missed_at = datetime('now')`
            : '';
      db.prepare(`UPDATE appointments SET status = ?, updated_at = datetime('now')${extra} WHERE id = ?`).run(status, appointment.id);

      if (status === 'missed' && notifications) {
        notifications.notifyRole('administrator', {
          type: 'missed_appointment',
          title: 'Missed appointment',
          message: `${appointment.firstName} ${appointment.lastName} missed ${appointment.examination} on ${appointment.date}.`,
          resourceType: 'appointment',
          resourceId: appointment.id,
        });
        alerts?.raise({
          type: 'no_show',
          description: `Missed appointment ${appointment.id} for ${appointment.firstName} ${appointment.lastName}.`,
          resource: `appointment:${appointment.id}`,
          dedupeMinutes: 60,
        });
      }

      if (status === 'cancelled' && notifications) {
        notifications.notifyRole('administrator', {
          type: 'cancelled_appointment',
          title: 'Appointment cancelled',
          message: `Appointment ${appointment.id} (${appointment.examination}) was cancelled.`,
          resourceType: 'appointment',
          resourceId: appointment.id,
        });
      }

      void actorId;
      return get(appointment.id);
    },

    processOverdue() {
      const today = localDate();
      const overdue = db
        .prepare(
          `SELECT id FROM appointments
           WHERE status = 'scheduled' AND (scheduled_date < @today OR (scheduled_date = @today AND scheduled_time < @now))`,
        )
        .all({ today, now: new Date().toTimeString().slice(0, 5) });

      for (const row of overdue) {
        this.setStatus(row.id, 'missed', null);
      }

      const graceDue = db
        .prepare(
          `SELECT a.id, p.first_name AS firstName, p.last_name AS lastName, a.examination, a.scheduled_date AS date
           FROM appointments a
           JOIN patients p ON p.id = a.patient_id
           WHERE a.status = 'missed'
             AND a.grace_alert_sent = 0
             AND datetime(a.scheduled_date || ' ' || a.scheduled_time) <= datetime('now', '-2 days')`,
        )
        .all();

      for (const row of graceDue) {
        notifications?.notifyRole('administrator', {
          type: 'unattended_grace',
          title: 'Unattended appointment (2-day grace elapsed)',
          message: `${row.firstName} ${row.lastName}: ${row.examination} on ${row.date} remains unattended after the two-day grace period.`,
          resourceType: 'appointment',
          resourceId: row.id,
        });
        alerts?.raise({
          type: 'no_show',
          description: `Two-day grace elapsed for unattended appointment ${row.id} (${row.firstName} ${row.lastName}).`,
          resource: `appointment-grace:${row.id}`,
        });
        db.prepare(`UPDATE appointments SET grace_alert_sent = 1 WHERE id = ?`).run(row.id);
      }

      return { markedMissed: overdue.length, graceAlerts: graceDue.length };
    },
  };
}

function upsertUser(db, { username, password, role, fullName, department }) {
  const existing = db.prepare('SELECT id FROM users WHERE username = ?').get(username);

    .prepare(

       VALUES (?, ?, ?, ?, ?, 'active', 0, datetime('now'))`,
    )
    .run(username, bcrypt.hashSync(password, 10), role, fullName, department);

}

function handle(fn) {
  return async (req, res, next) => {
    try {
      await fn(req, res);
    } catch (error) {
      if (error instanceof UserError || error instanceof ScheduleError) {
        return res.status(error.status).json({ error: error.message });
      }
      return next(error);
    }

function requestDate(value) {
  return isValidDate(value) ? value : localDate();
}

function shiftDate(date, days) {
  const value = new Date(`${date}T12:00:00`);
  value.setDate(value.getDate() + days);
  return localDate(value);
}

function roleLabel(role) {
  if (role === 'administrator') return 'Administrator';
  if (role === 'radiologist') return 'Radiologist';
  return 'Practitioner';
}

function splitModalities(value) {
  return String(value || '')
    .split(/[\\,]\s*/)
    .map((item) => item.trim().toUpperCase())
    .filter(Boolean);
}

export class AppError extends Error {
  constructor(message, status = 400) {
    super(message);
    this.status = status;
  }
}

export function handleRoute(fn) {
  return async (req, res, next) => {
    try {
      await fn(req, res);
    } catch (error) {
      if (error.status) {
        return res.status(error.status).json({ error: error.message });
      }
      return next(error);
    }
  };
}

export function cleanText(value, maxLength = 240) {
  if (value === undefined) {
    return undefined;
  }
  const text = String(value ?? '').trim();
  return text ? text.slice(0, maxLength) : null;
}

export function requireFields(body, fields) {
  for (const field of fields) {
    if (!String(body?.[field] ?? '').trim()) {
      throw new AppError(`${field} is required.`);
    }
  }
}

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';
import api from '../api/client.js';

const AuthContext = createContext(null);
const TOKEN_KEY = 'orthoviewplus_token';
const USER_KEY = 'orthoviewplus_user';

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [token, setToken] = useState(null);
  const [ready, setReady] = useState(false);

  const clearSession = useCallback(() => {
    localStorage.removeItem(TOKEN_KEY);
    localStorage.removeItem(USER_KEY);
    setToken(null);
    setUser(null);
  }, []);

  useEffect(() => {
    const interceptor = api.interceptors.response.use(
      (response) => response,
      (error) => {
        if (error.response?.status === 401 && localStorage.getItem(TOKEN_KEY)) {
          clearSession();
        }

        return Promise.reject(error);
      },
    );

    return () => {
      api.interceptors.response.eject(interceptor);
    };
  }, [clearSession]);

  useEffect(() => {
    let cancelled = false;

    async function restoreSession() {
      const storedToken = localStorage.getItem(TOKEN_KEY);
      const storedUser = localStorage.getItem(USER_KEY);

      if (!storedToken) {
        if (!cancelled) {
          setReady(true);
        }
        return;
      }

      if (storedUser) {
        try {
          setUser(JSON.parse(storedUser));
          setToken(storedToken);
        } catch {
          clearSession();
        }
      } else {
        setToken(storedToken);
      }

      try {
        const response = await api.get('/auth/me');
        if (cancelled) {
          return;
        }

        localStorage.setItem(USER_KEY, JSON.stringify(response.data.user));
        setUser(response.data.user);
        setToken(storedToken);
      } catch {
        if (!cancelled) {
          clearSession();
        }
      } finally {
        if (!cancelled) {
          setReady(true);
        }
      }
    }

    restoreSession();

    return () => {
      cancelled = true;
    };
  }, [clearSession]);

  const value = useMemo(
    () => ({
      user,
      token,
      ready,
      isAuthenticated: Boolean(token),
      async login(username, password) {
        const response = await api.post('/auth/login', { username, password });
        localStorage.setItem(TOKEN_KEY, response.data.token);
        localStorage.setItem(USER_KEY, JSON.stringify(response.data.user));
        setToken(response.data.token);
        setUser(response.data.user);
        return response.data.user;
      },
      async logout() {
        try {
          await api.post('/auth/logout');
        } catch {
          // Local session is still cleared even if the audit request fails.
        }
        clearSession();
      },
    }),
    [user, token, ready, clearSession],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within AuthProvider');
  }
  return context;
}

    async function loadStudies() {
      setLoadingStudies(true);
      setError('');
      try {
        const response = await api.get('/compare/studies');
        if (!cancelled) {
          setStudies(response.data.studies || []);
        }
      } catch (err) {
        if (!cancelled) {
          setError(err.response?.data?.error || 'Unable to load studies for comparison.');
          setStudies([]);
        }
      } finally {
        if (!cancelled) {
          setLoadingStudies(false);
        }
      }
    }

    loadStudies();
    return () => {
      cancelled = true;
    };
  }, []);

  const filteredStudies = useMemo(() => {
    const needle = studySearch.trim().toLowerCase();
    if (!needle) {
      return studies;
    }

    return studies.filter((study) =>
      [study.studyDate, study.studyDescription, study.patientId, study.studyId, study.modalitiesInStudy]
        .filter(Boolean)
        .some((value) => String(value).toLowerCase().includes(needle)),
    );
  }, [studySearch, studies]);

  async function handleCompare() {
    if (!selectedLeft || !selectedRight || selectedLeft === selectedRight) {
      setError('Select two different studies to compare.');
      return;
    }

    setLoadingComparison(true);
    setError('');
    try {
      const response = await api.get('/compare/compare', {
        params: {
          leftStudyId: selectedLeft,
          rightStudyId: selectedRight,
        },
      });
      setComparison(response.data.comparison);
    } catch (err) {
      setComparison(null);
      setError(err.response?.data?.error || 'Unable to compare the selected studies.');
    } finally {
      setLoadingComparison(false);
    }
  }

  const leftProfile = comparison?.left;
  const rightProfile = comparison?.right;

  return (
    <section className="page">
      {loadingStudies ? <LoadingState message="Loading available studies..." /> : null}
      <ErrorMessage message={error} />

      <section className="panel">
        <div className="panel-header">
          <div className="section-heading">
            <h3>Study Selection</h3>
            <p>{studies.length} studies available from Orthanc</p>
          </div>
        </div>
        <div className="toolbar comparison-controls">
          <div className="field inline-field">
            <label htmlFor="study-search">Search studies</label>
            <div className="input-icon">
              <SearchIcon size={16} />
              <input
                id="study-search"
                value={studySearch}
                onChange={(event) => setStudySearch(event.target.value)}
                placeholder="Date, description or patient ID"
              />
            </div>
          </div>
          <div className="field inline-field">
            <label htmlFor="left-study">
              <span className="series-tag">A</span> First study
            </label>
            <select id="left-study" value={selectedLeft} onChange={(event) => setSelectedLeft(event.target.value)}>
              <option value="">Select a study</option>
              {filteredStudies.map((study) => (
                <StudyOption key={study.id} study={study} />
              ))}
            </select>
          </div>
          <div className="field inline-field">
            <label htmlFor="right-study">
              <span className="series-tag">B</span> Second study
            </label>
            <select id="right-study" value={selectedRight} onChange={(event) => setSelectedRight(event.target.value)}>
              <option value="">Select a study</option>
              {filteredStudies.map((study) => (
                <StudyOption key={study.id} study={study} />
              ))}
            </select>
          </div>
          <button type="button" className="primary-button" onClick={handleCompare} disabled={loadingComparison}>
            <CompareIcon size={16} />
            <span>{loadingComparison ? 'Comparing...' : 'Compare Studies'}</span>
          </button>
        </div>
      </section>

      <div className="viewport-grid">
        <StudyViewport label="A" profile={leftProfile} />
        <StudyViewport label="B" profile={rightProfile} />
      </div>

      {comparison ? (
        <section className="panel">
          <div className="panel-header">
            <div className="section-heading">
              <h3>Metadata Comparison</h3>
              <p>No clinical conclusion is generated. Missing values are shown as unavailable.</p>
            </div>
            <span className="legend">
              <span className="diff-chip">Differs</span> Values differ between studies
            </span>
          </div>
          <div className="table-wrap flush">
            <table className="data-table comparison-table">
              <thead>
                <tr>
                  <th>Field</th>
                  <th>
                    <span className="series-tag">A</span> Study A
                  </th>
                  <th>
                    <span className="series-tag">B</span> Study B
                  </th>
                  <th className="diff-col" aria-label="Difference" />
                </tr>
              </thead>
              <tbody>
                {comparisonGroups.map((group) => (
                  <Fragment key={group.title}>
                    <tr className="group-row">
                      <th colSpan={4} scope="colgroup">
                        {group.title}
                      </th>
                    </tr>
                    {group.fields.map((field) => {
                      const left = field.getValue(leftProfile);
                      const right = field.getValue(rightProfile);
                      const differs = isDifferent(left, right);
                      return (
                        <tr key={field.label} className={differs ? 'row-diff' : undefined}>
                          <th scope="row">{field.label}</th>
                          <td>{formatValue(left)}</td>
                          <td>{formatValue(right)}</td>
                          <td className="diff-col">{differs ? <span className="diff-chip">Differs</span> : null}</td>
                        </tr>
                      );
                    })}
                  </Fragment>
                ))}
              </tbody>
            </table>
          </div>
        </section>
      ) : (
        <EmptyState
          title="No comparison loaded"
          message="Select two studies, then run the comparison to review side-by-side metadata."
        />
      )}
    </section>
  );
}

from __future__ import annotations

import io
from typing import Any

import cv2
import numpy as np
from PIL import Image

try:
    import pydicom
except ImportError:  # pragma: no cover
    pydicom = None


def decode_upload(data: bytes) -> tuple[np.ndarray, dict[str, Any]]:
    """Decode DICOM bytes or a standard image into an 8-bit grayscale array."""
    metadata: dict[str, Any] = {"format": "image"}
    if pydicom is not None:
        try:
            dataset = pydicom.dcmread(io.BytesIO(data), force=True)
            if hasattr(dataset, "pixel_array"):
                pixels = dataset.pixel_array.astype(np.float32)
                slope = float(getattr(dataset, "RescaleSlope", 1) or 1)
                intercept = float(getattr(dataset, "RescaleIntercept", 0) or 0)
                pixels = pixels * slope + intercept
                display = _window_array(pixels, dataset)
                metadata = {
                    "format": "dicom",
                    "patient_id": str(getattr(dataset, "PatientID", "") or ""),
                    "study_date": str(getattr(dataset, "StudyDate", "") or ""),
                    "modality": str(getattr(dataset, "Modality", "") or ""),
                    "sop_instance_uid": str(getattr(dataset, "SOPInstanceUID", "") or ""),
                    "study_instance_uid": str(getattr(dataset, "StudyInstanceUID", "") or ""),
                    "rows": int(display.shape[0]),
                    "columns": int(display.shape[1]),
                    "photometric": str(getattr(dataset, "PhotometricInterpretation", "") or ""),
                }
                return display, metadata
        except Exception:
            pass

    encoded = np.frombuffer(data, dtype=np.uint8)
    image = cv2.imdecode(encoded, cv2.IMREAD_GRAYSCALE)
    if image is None:
        pil = Image.open(io.BytesIO(data)).convert("L")
        image = np.array(pil)
    metadata["rows"] = int(image.shape[0])
    metadata["columns"] = int(image.shape[1])
    return image, metadata


def _window_array(pixels: np.ndarray, dataset) -> np.ndarray:
    center = getattr(dataset, "WindowCenter", None)
    width = getattr(dataset, "WindowWidth", None)
    if center is not None and width is not None:
        try:
            center = float(center[0] if hasattr(center, "__iter__") else center)
            width = float(width[0] if hasattr(width, "__iter__") else width)
            low = center - width / 2.0
            high = center + width / 2.0
            scaled = np.clip((pixels - low) / max(high - low, 1e-6), 0, 1)
            return (scaled * 255).astype(np.uint8)
        except Exception:
            pass
    low, high = np.percentile(pixels, (1, 99))
    scaled = np.clip((pixels - low) / max(high - low, 1e-6), 0, 1)
    return (scaled * 255).astype(np.uint8)


def encode_png(image: np.ndarray) -> bytes:
    if image.ndim == 2:
        ok, buffer = cv2.imencode(".png", image)
    else:
        ok, buffer = cv2.imencode(".png", cv2.cvtColor(image, cv2.COLOR_RGB2BGR))
    if not ok:
        raise ValueError("Unable to encode PNG output.")
    return buffer.tobytes()


def to_base64(image: np.ndarray) -> str:
    import base64

    return base64.b64encode(encode_png(image)).decode("ascii")


def overlay_mask(base: np.ndarray, mask: np.ndarray, color=(220, 70, 70), alpha=0.45) -> np.ndarray:
    rgb = cv2.cvtColor(base, cv2.COLOR_GRAY2RGB)
    tint = np.zeros_like(rgb)
    tint[:, :] = color
    binary = (mask > 0).astype(np.uint8)
    for channel in range(3):
        rgb[:, :, channel] = np.where(
            binary == 1,
            (alpha * tint[:, :, channel] + (1 - alpha) * rgb[:, :, channel]).astype(np.uint8),
            rgb[:, :, channel],
        )
    return rgb
