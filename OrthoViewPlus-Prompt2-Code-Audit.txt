router = APIRouter()


async def read_required(upload: UploadFile | None, name: str) -> bytes:
    if upload is None:
        raise HTTPException(status_code=400, detail=f"{name} is required.")
    data = await upload.read()
    if not data:
        raise HTTPException(status_code=400, detail=f"{name} is empty.")
    return data


@router.post("/restore")
async def restore(image: UploadFile = File(...)):
    return restore_image(await read_required(image, "image"))


@router.post("/completeness-check")
async def completeness(image: UploadFile = File(...), exam_type: str = Form(""), body_part: str = Form("")):
    return assess_completeness(await read_required(image, "image"), exam_type or body_part)


@router.post("/compare")
async def compare(

    prior_image: UploadFile = File(...),
    patient_id: str = Form(""),

):
    return compare_images(

from __future__ import annotations

import cv2
import numpy as np

from .io_utils import decode_upload, to_base64


METHOD = {
    "name": "OpenCV non-local means + CLAHE restoration pipeline",
    "library": "opencv-python-headless",
    "licence": "Apache-2.0 (OpenCV)",
    "notes": (
        "Classical restoration: denoise with fastNlMeansDenoising, local contrast with CLAHE, "
        "and gentle unsharp masking. This is not a diagnostic enhancement model and does not invent anatomy."
    ),
}


def restore_image(data: bytes) -> dict:
    image, metadata = decode_upload(data)
    denoised = cv2.fastNlMeansDenoising(image, None, h=8, templateWindowSize=7, searchWindowSize=21)
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
    equalised = clahe.apply(denoised)
    blur = cv2.GaussianBlur(equalised, (0, 0), 1.0)
    restored = cv2.addWeighted(equalised, 1.25, blur, -0.25, 0)
    restored = np.clip(restored, 0, 255).astype(np.uint8)

    residual = cv2.absdiff(image, restored)
    metrics = {
        "mean_absolute_change": float(np.mean(residual)),
        "noise_estimate_original": float(np.mean(cv2.Laplacian(image, cv2.CV_64F) ** 2)),
        "noise_estimate_restored": float(np.mean(cv2.Laplacian(restored, cv2.CV_64F) ** 2)),
    }
    return {
        "result_kind": "restored_image",
        "method": METHOD,
        "licence": METHOD["licence"],
        "metrics": metrics,
        "source_metadata": metadata,
        "warnings": [
            "The original acquisition is unchanged.",
            "Restoration may reduce noise but must not be treated as a new diagnostic image.",
            "Review the original image alongside this output.",
        ],
        "label": "AI-restored image — derived, requires radiologist review",
        "image_base64": to_base64(restored),
        "experimental": False,
        "synthetic": False,
    }

from __future__ import annotations

import cv2
import numpy as np

from .io_utils import decode_upload, overlay_mask, to_base64

METHOD = {
    "name": "Field-of-view coverage heuristic (content bounding box vs frame)",
    "library": "OpenCV",
    "licence": "Apache-2.0 (OpenCV)",
    "notes": (
        "Estimates whether anatomy is cropped by comparing the content bounding box with the image frame "
        "and flagging large empty margins or content touching the border."
    ),
}


def assess_completeness(data: bytes, exam_type: str = "") -> dict:
    image, metadata = decode_upload(data)
    blurred = cv2.GaussianBlur(image, (5, 5), 0)
    _, binary = cv2.threshold(blurred, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    if np.mean(binary) > 127:
        binary = 255 - binary
    kernel = np.ones((9, 9), np.uint8)
    binary = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, kernel)
    coords = cv2.findNonZero(binary)
    h, w = image.shape[:2]
    missing_sides = []
    coverage_ratio = 0.0
    mask = np.zeros_like(image)
    if coords is not None:
        x, y, bw, bh = cv2.boundingRect(coords)
        coverage_ratio = (bw * bh) / float(w * h)
        margin = {"left": x / w, "right": (w - (x + bw)) / w, "top": y / h, "bottom": (h - (y + bh)) / h}
        touches = {
            "left": x <= 2,
            "right": x + bw >= w - 2,
            "top": y <= 2,
            "bottom": y + bh >= h - 2,
        }
        for side, ratio in margin.items():
            if ratio > 0.12 and not touches[side]:
                missing_sides.append(side)
            if touches[side] and ratio < 0.01:
                missing_sides.append(f"{side}-border-cut")
        mask[y : y + bh, x : x + bw] = 255
        cv2.rectangle(mask, (x, y), (x + bw, y + bh), 180, 2)
    else:
        missing_sides = ["no-anatomy-detected"]

    exam = (exam_type or "").lower()
    exam_notes = []
    if "foot" in exam or "toe" in exam:
        exam_notes.append("For foot examinations, confirm all toes are included at the distal border.")
    if "chest" in exam:
        exam_notes.append("For chest examinations, confirm lung apices and costophrenic angles are included.")

    overlay = overlay_mask(image, 255 - mask if missing_sides else mask, color=(229, 162, 59), alpha=0.35)
    return {
        "result_kind": "completeness_assessment",
        "method": METHOD,
        "licence": METHOD["licence"],
        "metrics": {
            "coverage_ratio": round(float(coverage_ratio), 4),
            "possible_missing_coverage": sorted(set(missing_sides)),
        },
        "source_metadata": metadata,
        "warnings": [
            "This is an assistive coverage check, not a determination that the examination is diagnostically adequate.",
            *exam_notes,
        ],
        "label": "Coverage assessment — assistive, requires review",
        "image_base64": to_base64(overlay),
        "experimental": False,
        "synthetic": False,
    }

from __future__ import annotations

import cv2
import numpy as np

from .io_utils import decode_upload, to_base64

METHOD = {
    "name": "ECC affine alignment with absolute-difference heatmap",
    "library": "OpenCV findTransformECC",
    "licence": "Apache-2.0 (OpenCV)",
}


def compare_images(current_bytes: bytes, prior_bytes: bytes, patient_id: str = "", prior_patient_id: str = "") -> dict:
    current, current_meta = decode_upload(current_bytes)
    prior, prior_meta = decode_upload(prior_bytes)

    current_pid = patient_id or current_meta.get("patient_id") or ""
    prior_pid = prior_patient_id or prior_meta.get("patient_id") or ""
    if current_pid and prior_pid and current_pid != prior_pid:
        return {
            "result_kind": "comparison_rejected",
            "method": METHOD,
            "licence": METHOD["licence"],
            "metrics": {},
            "warnings": ["Refused: Patient IDs do not match. Unrelated patients cannot be compared."],
            "label": "Comparison rejected — patient identity mismatch",
            "experimental": False,
            "synthetic": False,
        }

    prior_resized = cv2.resize(prior, (current.shape[1], current.shape[0]), interpolation=cv2.INTER_AREA)
    warp = np.eye(2, 3, dtype=np.float32)
    aligned = prior_resized
    alignment = "none"
    try:
        criteria = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 80, 1e-5)
        _, warp = cv2.findTransformECC(
            current.astype(np.float32) / 255.0,
            prior_resized.astype(np.float32) / 255.0,
            warp,
            cv2.MOTION_AFFINE,
            criteria,
            None,
            5,
        )
        aligned = cv2.warpAffine(prior_resized, warp, (current.shape[1], current.shape[0]), flags=cv2.INTER_LINEAR + cv2.WARP_INVERSE_MAP)
        alignment = "ecc-affine"
    except cv2.error:
        alignment = "resize-only"

    diff = cv2.absdiff(current, aligned)
    heatmap = cv2.applyColorMap(diff, cv2.COLORMAP_INFERNO)
    current_rgb = cv2.cvtColor(current, cv2.COLOR_GRAY2BGR)
    blended = cv2.addWeighted(current_rgb, 0.55, heatmap, 0.45, 0)

    return {
        "result_kind": "longitudinal_comparison",
        "method": METHOD,
        "licence": METHOD["licence"],
        "metrics": {
            "mean_absolute_difference": float(np.mean(diff)),
            "alignment": alignment,
            "current_study_date": current_meta.get("study_date"),
            "prior_study_date": prior_meta.get("study_date"),
        },
        "source_metadata": {"current": current_meta, "prior": prior_meta},
        "warnings": [
            "Highlighted regions indicate intensity change after alignment, not a diagnosis.",
            "Always inspect the original earlier and later acquisitions.",
        ],
        "label": "Longitudinal comparison — difference map, requires review",
        "image_base64": to_base64(cv2.cvtColor(blended, cv2.COLOR_BGR2RGB)),
        "overlay_base64": to_base64(aligned),
        "experimental": False,
        "synthetic": False,
    }

from __future__ import annotations

import cv2
import numpy as np

from .io_utils import decode_upload, overlay_mask, to_base64

METHOD = {
    "name": "Prior-exam aligned inpainting of low-signal regions",
    "library": "OpenCV inpaint + ECC alignment",
    "licence": "Apache-2.0 (OpenCV)",
    "notes": "Estimated anatomy from a prior examination of the same patient. Not exact recovery.",
}


def reconstruct(current_bytes: bytes, prior_bytes: bytes, patient_id: str = "", prior_patient_id: str = "") -> dict:
    current, current_meta = decode_upload(current_bytes)
    prior, prior_meta = decode_upload(prior_bytes)
    current_pid = patient_id or current_meta.get("patient_id") or ""
    prior_pid = prior_patient_id or prior_meta.get("patient_id") or ""
    if current_pid and prior_pid and current_pid != prior_pid:
        return {
            "result_kind": "reconstruction_rejected",
            "method": METHOD,
            "licence": METHOD["licence"],
            "metrics": {},
            "warnings": ["Refused: prior and current Patient IDs do not match."],
            "label": "Reconstruction rejected — patient identity mismatch",
            "experimental": True,
            "synthetic": False,
        }

    prior_resized = cv2.resize(prior, (current.shape[1], current.shape[0]))
    warp = np.eye(2, 3, dtype=np.float32)
    try:
        _, warp = cv2.findTransformECC(
            current.astype(np.float32) / 255.0,
            prior_resized.astype(np.float32) / 255.0,
            warp,
            cv2.MOTION_AFFINE,
            (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 60, 1e-4),
            None,
            5,
        )
        aligned = cv2.warpAffine(prior_resized, warp, (current.shape[1], current.shape[0]), flags=cv2.INTER_LINEAR + cv2.WARP_INVERSE_MAP)
    except cv2.error:
        aligned = prior_resized

    missing = (current < 12).astype(np.uint8) * 255
    missing = cv2.morphologyEx(missing, cv2.MORPH_CLOSE, np.ones((11, 11), np.uint8))
    reconstructed = current.copy()
    reconstructed[missing > 0] = aligned[missing > 0]
    reconstructed = cv2.inpaint(reconstructed, missing, 3, cv2.INPAINT_TELEA)
    # Preserve brighter (likely abnormal or true anatomy) pixels from the current image.
    reconstructed = np.where(current > 20, current, reconstructed).astype(np.uint8)
    overlay = overlay_mask(reconstructed, missing, color=(21, 154, 145), alpha=0.4)

    return {
        "result_kind": "anatomical_reconstruction",
        "method": METHOD,
        "licence": METHOD["licence"],
        "metrics": {
            "estimated_pixel_fraction": float(np.mean(missing > 0)),
        },
        "source_metadata": {"current": current_meta, "prior": prior_meta},
        "warnings": [
            "Highlighted pixels are estimated from a prior examination, not recovered from the current acquisition.",
            "Visible abnormalities in the current image are preserved where signal is present.",
            "Do not treat this as an exact anatomical recovery.",
        ],
        "label": "Estimated reconstruction — generated regions identified on overlay",
        "image_base64": to_base64(reconstructed),
        "overlay_base64": to_base64(overlay),
        "experimental": True,
        "synthetic": False,
    }

from __future__ import annotations

import numpy as np
from skimage import measure

from .io_utils import decode_upload

METHOD = {
    "name": "Intensity isosurface (marching cubes) on stacked slices",
    "library": "scikit-image.measure.marching_cubes",
    "licence": "BSD-3-Clause (scikit-image)",
    "notes": "Intended for CT volumes. A single 2D radiograph cannot produce accurate 3D anatomy.",
}


def segment_volume(files: list[bytes]) -> dict:
    if len(files) < 8:
        return {
            "result_kind": "segmentation_rejected",
            "method": METHOD,
            "licence": METHOD["licence"],
            "metrics": {"slice_count": len(files)},
            "warnings": [
                "A CT volume with multiple slices is required.",
                "A single ordinary 2D X-ray does not provide an accurate 3D reconstruction of anatomy.",
            ],
            "label": "3D visualisation unavailable — insufficient volume data",
            "experimental": False,
            "synthetic": False,
        }

    slices = []
    modalities = set()
    for item in files:
        image, metadata = decode_upload(item)
        slices.append(image)
        if metadata.get("modality"):
            modalities.add(metadata["modality"])
    if modalities and not any(mod in {"CT", "MR"} for mod in modalities):
        return {
            "result_kind": "segmentation_rejected",
            "method": METHOD,
            "licence": METHOD["licence"],
            "metrics": {"modalities": sorted(modalities)},
            "warnings": ["3D isosurface visualisation is supported for CT/MR volumes, not projection radiographs."],
            "label": "3D visualisation unavailable — unsupported modality",
            "experimental": False,
            "synthetic": False,
        }

    volume = np.stack(slices, axis=0).astype(np.float32)
    volume = volume[::2, ::2, ::2]
    threshold = float(np.percentile(volume, 85))
    try:
        verts, faces, _normals, _values = measure.marching_cubes(volume, level=threshold, step_size=2)
    except Exception as error:
        return {
            "result_kind": "segmentation_failed",
            "method": METHOD,
            "licence": METHOD["licence"],
            "metrics": {},
            "warnings": [f"Marching cubes failed: {error}"],
            "label": "3D visualisation failed",
            "experimental": False,
            "synthetic": False,
        }

    verts = verts.astype(np.float32)
    verts -= verts.mean(axis=0)
    scale = np.max(np.linalg.norm(verts, axis=1)) or 1.0
    verts /= scale
    # Cap mesh size for browser delivery.
    faces = faces[:25000]
    return {
        "result_kind": "anatomical_3d",
        "method": METHOD,
        "licence": METHOD["licence"],
        "metrics": {
            "slices_used": len(slices),
            "threshold_percentile": 85,
            "vertex_count": int(len(verts)),
            "face_count": int(len(faces)),
        },
        "warnings": [
            "This is a density isosurface, not a validated anatomical segmentation.",
            "Rotate and inspect the mesh, then compare with the original slices.",
        ],
        "label": "3D isosurface derived from volume — requires radiologist review",
        "mesh": {
            "vertices": verts.tolist(),
            "faces": faces.astype(int).tolist(),
        },
        "experimental": False,
        "synthetic": False,
    }

from __future__ import annotations

import numpy as np

from .io_utils import to_base64

METHOD = {
    "name": "Procedural radiographic phantom generator",
    "library": "NumPy",
    "licence": "Project code (academic prototype)",
    "notes": "No patient data is used. Output is synthetic and stored separately from clinical studies.",
}


def generate_synthetic(seed: int = 7) -> dict:
    rng = np.random.default_rng(int(seed))
    height, width = 512, 512
    yy, xx = np.mgrid[0:height, 0:width]
    image = np.full((height, width), 18.0)
    # Soft-tissue torso ellipse
    torso = ((xx - 256) ** 2) / (170**2) + ((yy - 270) ** 2) / (220**2)
    image[torso < 1] = 70
    # Lung fields
    for cx in (190, 322):
        lungs = ((xx - cx) ** 2) / (70**2) + ((yy - 240) ** 2) / (120**2)
        image[lungs < 1] = 32
    # Spine
    image[np.abs(xx - 256) < 12] += 40
    # Rib arcs
    for radius in range(70, 180, 16):
        dist = np.sqrt((xx - 256) ** 2 + (yy - 210) ** 2)
        rib = np.abs(dist - radius) < 2.2
        image[rib] += 28
    image += rng.normal(0, 4.5, image.shape)
    image = np.clip(image, 0, 255).astype(np.uint8)

    return {
        "result_kind": "synthetic_image",
        "method": METHOD,
        "licence": METHOD["licence"],
        "metrics": {"seed": int(seed), "width": width, "height": height},
        "warnings": [
            "SYNTHETIC IMAGE. Not a patient examination.",
            "Do not use this output to represent clinical findings.",
        ],
        "label": "SYNTHETIC — demonstration phantom, not a clinical study",
        "image_base64": to_base64(image),
        "experimental": False,
        "synthetic": True,
        "generation_process": {
            "type": "procedural_numpy_phantom",
            "components": ["torso ellipse", "lung fields", "spine", "rib arcs", "gaussian noise"],
            "seed": int(seed),
        },
    }

from __future__ import annotations

import cv2
import numpy as np

from .io_utils import decode_upload, to_base64

METHOD = {
    "name": "Experimental shear/reprojection estimate of a complementary view",
    "library": "OpenCV geometric transforms",
    "licence": "Apache-2.0 (OpenCV)",
    "notes": "Research prototype. Never a substitute for a required clinical projection.",
}


def complementary_view(data: bytes) -> dict:
    image, metadata = decode_upload(data)
    h, w = image.shape[:2]
    # Horizontal intensity projection expanded into an estimated orthogonal-looking image.
    profile = image.mean(axis=1)
    estimated = np.tile(profile.reshape(h, 1), (1, w))
    shear = np.float32([[1, 0.35, 0], [0, 1, 0]])
    warped = cv2.warpAffine(image, shear, (w, h), borderMode=cv2.BORDER_REPLICATE)
    mixed = cv2.addWeighted(estimated.astype(np.uint8), 0.45, warped, 0.55, 0)
    mixed = cv2.GaussianBlur(mixed, (0, 0), 1.2)

    return {
        "result_kind": "complementary_view",
        "method": METHOD,
        "licence": METHOD["licence"],
        "metrics": {"source_rows": h, "source_columns": w},
        "source_metadata": metadata,
        "warnings": [
            "SYNTHETIC / ESTIMATED VIEW. This is not an acquired projection.",
            "It must never replace a required clinical examination.",
            "This capability is experimental.",
        ],
        "label": "SYNTHETIC complementary view — estimated, not acquired",
        "image_base64": to_base64(mixed),
        "experimental": True,
        "synthetic": True,
    }

def _png_bytes():
    image = np.linspace(0, 255, 64 * 64, dtype=np.uint8).reshape(64, 64)

// Additive, idempotent migration: existing rows keep their data and default to an active, unrestricted account.
function migrateUsers(db) {
  const existing = new Set(db.prepare('PRAGMA table_info(users)').all().map((column) => column.name));
  const columns = [
    ['full_name', 'TEXT'],
    ['email', 'TEXT'],
    ['department', 'TEXT'],
    ['status', "TEXT NOT NULL DEFAULT 'active'"],
    ['restricted', 'INTEGER NOT NULL DEFAULT 0'],
    ['restriction_reason', 'TEXT'],
    ['updated_at', 'TEXT'],
    ['last_login_at', 'TEXT'],
  ];

  for (const [name, definition] of columns) {
    if (!existing.has(name)) {
      db.exec(`ALTER TABLE users ADD COLUMN ${name} ${definition}`);
    }
  }
}

import jwt from 'jsonwebtoken';

export function createAuthMiddleware(jwtSecret, { getAuthState } = {}) {
  return function requireAuth(req, res, next) {
    const header = req.headers.authorization || '';
    const [scheme, token] = header.split(' ');

    if (scheme !== 'Bearer' || !token) {
      return res.status(401).json({ error: 'Authentication required.' });
    }

    let payload;
    try {
      payload = jwt.verify(token, jwtSecret);
    } catch {
      return res.status(401).json({ error: 'Session expired or invalid. Please sign in again.' });
    }

    req.user = {
      id: payload.sub,
      username: payload.username,
      role: payload.role,
    };

    // Re-check the account on every request so deactivation, restriction and role changes apply immediately.
    if (getAuthState) {
      const account = getAuthState(payload.sub);
      if (!account || account.status !== 'active' || account.restricted) {
        return res.status(401).json({ error: 'This account is no longer permitted to sign in.' });
      }
      req.user = { id: account.id, username: account.username, role: account.role };
    }

    return next();
  };
}

export function createRoleMiddleware(allowedRoles) {
  const roles = Array.isArray(allowedRoles) ? allowedRoles : [allowedRoles];

  return function requireRole(req, res, next) {
    if (!req.user) {
      return res.status(401).json({ error: 'Authentication required.' });
    }

    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'You do not have permission to perform this action.' });
    }

    return next();
  };
}

import { Router } from 'express';
import { handleRoute } from '../utils/http.js';

export function createClinicalRouter({
  patients,
  appointments,
  requests,
  reports,
  studies,
  notifications,
  audit,
  users,
}) {
  const router = Router();

  function log(req, action, target, description) {
    audit.log({
      userId: req.user?.id ?? null,
      action,
      target,
      description: `User "${req.user?.username ?? 'unknown'}" ${description}`,
    });
  }

  router.get(
    '/patients',
    handleRoute((req, res) => {
      res.json({ patients: patients.search({ query: req.query.q, status: req.query.status || 'active' }) });
    }),
  );

  router.post(
    '/patients',
    handleRoute((req, res) => {
      if (!['administrator', 'practitioner'].includes(req.user.role)) {
        return res.status(403).json({ error: 'You do not have permission to register patients.' });
      }
      const patient = patients.create({ ...req.body, createdBy: req.user.id });
      log(req, 'patient_create', `patient:${patient.id}`, `registered patient ${patient.mrn}.`);
      res.status(201).json({ patient });
    }),
  );

  router.get(
    '/patients/:id',
    handleRoute((req, res) => {
      const patient = patients.requirePatient(req.params.id);
      const history = {
        appointments: appointments.list({ patientId: patient.id }),
        requests: requests.list({ patientId: patient.id }),
        studies: studies.list({ patientId: patient.id, source: 'all' }),
        reports: reports.list({
          patientId: patient.id,
          status: req.user.role === 'practitioner' ? 'approved' : 'all',
        }),
      };
      log(req, 'patient_view', `patient:${patient.id}`, `viewed patient ${patient.mrn}.`);
      res.json({ patient, history });
    }),
  );

  router.patch(
    '/patients/:id',
    handleRoute((req, res) => {
      if (req.user.role !== 'administrator') {
        return res.status(403).json({ error: 'You do not have permission to edit patient records.' });
      }
      const patient = patients.update(req.params.id, req.body || {});
      log(req, 'patient_update', `patient:${patient.id}`, `updated patient ${patient.mrn}.`);
      res.json({ patient });
    }),
  );

  router.get(
    '/appointments',
    handleRoute((req, res) => {
      res.json({
        appointments: appointments.list({
          status: req.query.status,
          dateFrom: req.query.dateFrom,
          dateTo: req.query.dateTo,
          patientId: req.query.patientId,
        }),
        counts: appointments.counts(),
      });
    }),
  );

  router.post(
    '/appointments',
    handleRoute((req, res) => {
      if (req.user.role !== 'administrator' && req.user.role !== 'practitioner') {
        return res.status(403).json({ error: 'You do not have permission to schedule appointments.' });
      }
      const appointment = appointments.create({
        ...req.body,
        createdBy: req.user.id,
        requestedBy: req.body.requestedBy || req.user.id,
      });
      log(req, 'appointment_create', `appointment:${appointment.id}`, `scheduled ${appointment.examination}.`);
      res.status(201).json({ appointment });
    }),
  );

  router.post(
    '/appointments/:id/status',
    handleRoute((req, res) => {
      if (req.user.role !== 'administrator') {
        return res.status(403).json({ error: 'Only administrators can change appointment status.' });
      }
      const appointment = appointments.setStatus(req.params.id, req.body?.status, req.user.id);
      log(req, 'appointment_status', `appointment:${appointment.id}`, `set appointment status to ${appointment.status}.`);
      res.json({ appointment });
    }),
  );

  router.get(
    '/requests',
    handleRoute((req, res) => {
      const filters = {
        status: req.query.status,
        patientId: req.query.patientId,
      };
      if (req.user.role === 'practitioner') {
        filters.requestedBy = req.user.id;
      }
      if (req.user.role === 'radiologist' && req.query.mine === 'true') {
        filters.assignedRadiologistId = req.user.id;
      }
      res.json({ requests: requests.list(filters) });
    }),
  );

  router.post(
    '/requests',
    handleRoute((req, res) => {
      if (req.user.role !== 'practitioner' && req.user.role !== 'administrator') {
        return res.status(403).json({ error: 'You do not have permission to submit imaging requests.' });
      }
      const request = requests.create({ ...req.body, requestedBy: req.user.id });
      log(req, 'imaging_request_create', `request:${request.id}`, `submitted imaging request ${request.examination}.`);
      res.status(201).json({ request });
    }),
  );

  router.post(
    '/requests/:id/assign',
    handleRoute((req, res) => {
      if (req.user.role !== 'radiologist') {
        return res.status(403).json({ error: 'Only radiologists can claim examinations.' });
      }
      const request = requests.assign(req.params.id, req.user.id);
      log(req, 'imaging_request_assign', `request:${request.id}`, 'claimed an imaging request.');
      res.json({ request });
    }),
  );

  router.get(
    '/reports',
    handleRoute((req, res) => {
      const filters = { patientId: req.query.patientId, status: req.query.status };
      if (req.user.role === 'practitioner') {
        filters.status = 'approved';
      }
      if (req.user.role === 'radiologist' && req.query.mine === 'true') {
        filters.radiologistId = req.user.id;
      }
      res.json({ reports: reports.list(filters) });
    }),
  );

  router.post(
    '/reports',
    handleRoute((req, res) => {
      if (req.user.role !== 'radiologist') {
        return res.status(403).json({ error: 'Only radiologists can prepare reports.' });
      }
      const report = reports.saveDraft({ ...req.body, radiologistId: req.user.id });
      log(req, 'report_draft', `report:${report.id}`, 'saved a radiology report draft.');
      res.json({ report });
    }),
  );

  router.post(
    '/reports/:id/approve',
    handleRoute((req, res) => {
      if (req.user.role !== 'radiologist') {
        return res.status(403).json({ error: 'Only radiologists can approve reports.' });
      }
      const report = reports.approve(req.params.id, req.user.id);
      log(req, 'report_approve', `report:${report.id}`, `approved report for ${report.mrn}.`);
      res.json({ report });
    }),
  );

  router.get(
    '/notifications',
    handleRoute((req, res) => {
      res.json({
        notifications: notifications.listForUser(req.user.id, { unreadOnly: req.query.unread === 'true' }),
        unread: notifications.unreadCount(req.user.id),
      });
    }),
  );

  router.post(
    '/notifications/:id/read',
    handleRoute((req, res) => {
      res.json({ notification: notifications.markRead(req.params.id, req.user.id) });
    }),
  );

  router.post(
    '/notifications/read-all',
    handleRoute((req, res) => {
      notifications.markAllRead(req.user.id);
      res.json({ ok: true });
    }),
  );

  router.get(
    '/staff',
    handleRoute((_req, res) => {
      res.json({
        radiologists: users.list().filter((user) => user.role === 'radiologist' && user.status === 'active'),
        practitioners: users.list().filter((user) => user.role === 'practitioner' && user.status === 'active'),
      });
    }),
  );

  return router;
}

    handleRoute(async (req, res) => {
      const profile = await orthanc.getStudyProfile(req.params.id);
      const series = await orthanc.listSeriesForStudy(req.params.id);
      const matchedPatient = profile.patientId
        ? patients.search({ query: profile.patientId, status: 'all' }).find((row) => row.mrn.toLowerCase() === String(profile.patientId).toLowerCase())
        : null;
      if (matchedPatient) {
        studies.upsertOrthancStudy({
          patientId: matchedPatient.id,
          orthancStudyId: req.params.id,
          studyInstanceUid: profile.studyInstanceUID,
          studyDate: profile.studyDate,
          modality: profile.primarySeries?.modality,
          description: profile.studyDescription,
        });
      }
      log(req, 'view_study', `orthanc-study:${req.params.id}`, 'opened an imaging study.');
      res.json({ study: profile, series, linkedPatient: matchedPatient || null });
    }),
  );

  router.get(
    '/instances/:id/preview',
    handleRoute(async (req, res) => {
      const preview = await orthanc.getInstancePreview(req.params.id);
      res.setHeader('Content-Type', preview.contentType);
      res.setHeader('Cache-Control', 'private, max-age=60');
      res.send(preview.buffer);
    }),
  );

  router.get(
    '/instances/:id/rendered',
    handleRoute(async (req, res) => {
      const rendered = await orthanc.getInstanceRendered(req.params.id, {
        'window-center': req.query.windowCenter,
        'window-width': req.query.windowWidth,
      });
      res.setHeader('Content-Type', rendered.contentType);
      res.send(rendered.buffer);
    }),
  );

  router.get(
    '/instances/:id/tags',
    handleRoute(async (req, res) => {
      const tags = await orthanc.getInstanceTags(req.params.id);
      res.json({ tags });
    }),
  );

  return router;
}

import fs from 'node:fs';
import { Router } from 'express';
import { handleRoute } from '../utils/http.js';

export function createAiRouter({ aiJobs, aiClient }) {
  const router = Router();

  router.get(
    '/health',
    handleRoute(async (_req, res) => {
      res.json(await aiClient.health());
    }),
  );

  router.get(
    '/jobs',
    handleRoute((req, res) => {
      res.json({ jobs: aiJobs.list({ requestedBy: req.user.id }) });
    }),
  );

  router.get(
    '/jobs/:id',
    handleRoute((req, res) => {
      const job = aiJobs.getJob(req.params.id);
      if (!job) {
        return res.status(404).json({ error: 'AI job not found.' });
      }
      if (Number(job.requestedBy) !== Number(req.user.id) && req.user.role !== 'radiologist') {
        return res.status(403).json({ error: 'You do not have permission to view this job.' });
      }
      res.json({ job });
    }),
  );

  router.post(
    '/jobs',
    handleRoute((req, res) => {
      const job = aiJobs.enqueue({
        capability: req.body?.capability,
        requestedBy: req.user.id,
        patientId: req.body?.patientId,
        sourceStudyId: req.body?.sourceStudyId,
        input: req.body?.input || {},
        auditUser: req.user.id,
      });
      res.status(202).json({ job });
    }),
  );

  router.post(
    '/results/:id/review',
    handleRoute((req, res) => {
      res.json({ result: aiJobs.markReviewed(req.params.id, req.user.id) });
    }),
  );

  function sendStoredFile(res, filePath, fallbackName) {
    if (!filePath || !fs.existsSync(filePath)) {
      return res.status(404).json({ error: `${fallbackName} is not available.` });
    }
    if (filePath.endsWith('.json')) {
      res.setHeader('Content-Type', 'application/json');
    } else {
      res.setHeader('Content-Type', 'image/png');
    }
    return fs.createReadStream(filePath).pipe(res);
  }

  router.get(
    '/results/:id/file',
    handleRoute((req, res) => {
      const result = aiJobs.getResult(req.params.id);
      if (!result) {
        return res.status(404).json({ error: 'AI result not found.' });
      }
      return sendStoredFile(res, result.filePath, 'Result file');
    }),
  );

  router.get(
    '/results/:id/overlay',
    handleRoute((req, res) => {
      const result = aiJobs.getResult(req.params.id);
      if (!result) {
        return res.status(404).json({ error: 'AI result not found.' });
      }
      return sendStoredFile(res, result.overlayPath, 'Overlay');
    }),
  );

  return router;
}

import axios from 'axios';

export function createOrthancService({ url, username, password, timeoutMs = 2500 }) {
  const baseURL = String(url || '').replace(/\/$/, '');
  const client = axios.create({
    baseURL,
    timeout: timeoutMs,
    auth: username ? { username, password } : undefined,
    validateStatus: (status) => status >= 200 && status < 300,
  });

  function isAvailableError(error) {
    return !error?.response;
  }

  function readTag(source, tagName) {
    return source?.RequestedTags?.[tagName] ?? source?.MainDicomTags?.[tagName] ?? source?.[tagName] ?? null;
  }

  function normalizeValue(value) {
    if (value === null || value === undefined || value === '') {
      return null;
    }

    if (Array.isArray(value)) {
      return value.length > 0 ? value.join(', ') : null;
    }

    return value;
  }

  function formatActivity(change) {
    return {
      id: change.ID,
      type: change.ChangeType,
      resourceType: change.ResourceType,
      timestamp: change.Date,
      path: change.Path,
    };
  }

  async function request(method, path, data) {
    const response = await client.request({ method, url: path, data });
    return response.data;
  }

  async function getSystem() {
    return request('GET', '/system');
  }

  async function getStatistics() {
    return request('GET', '/statistics');
  }

  async function getChanges(limit = 10) {
    return request('GET', `/changes?limit=${limit}`);
  }

  async function findResources(level, options = {}) {
    return request('POST', '/tools/find', {
      Level: level,
      Expand: true,
      ...options,
    });
  }

  async function getStudy(studyId) {
    return request('GET', `/studies/${studyId}`);
  }

  async function getSeries(seriesId) {
    return request('GET', `/series/${seriesId}`);
  }

  async function getInstance(instanceId) {
    return request('GET', `/instances/${instanceId}`);
  }

  async function getInstanceTags(instanceId) {
    return request('GET', `/instances/${instanceId}/simplified-tags`);
  }

  async function getInstancePreview(instanceId) {
    const response = await client.request({
      method: 'GET',
      url: `/instances/${instanceId}/preview`,
      responseType: 'arraybuffer',
    });
    return {
      buffer: Buffer.from(response.data),
      contentType: response.headers['content-type'] || 'image/jpeg',
    };
  }

  async function getInstanceRendered(instanceId, params = {}) {
    const response = await client.request({
      method: 'GET',
      url: `/instances/${instanceId}/rendered`,
      params,
      responseType: 'arraybuffer',
    });
    return {
      buffer: Buffer.from(response.data),
      contentType: response.headers['content-type'] || 'image/png',
    };
  }

  async function getInstanceFile(instanceId) {
    const response = await client.request({
      method: 'GET',
      url: `/instances/${instanceId}/file`,
      responseType: 'arraybuffer',
      timeout: timeoutMs > 2500 ? timeoutMs : 30000,
    });
    return Buffer.from(response.data);
  }

  async function getStudyExpanded(studyId) {
    return request('GET', `/studies/${studyId}?expand`);
  }

  async function listSeriesForStudy(studyId) {
    const study = await getStudy(studyId);
    const seriesIds = Array.isArray(study.Series) ? study.Series : [];
    const series = await Promise.all(seriesIds.map((id) => getSeries(id).catch(() => null)));
    return series.filter(Boolean).map((item) => ({
      id: item.ID,
      modality: item.MainDicomTags?.Modality ?? null,
      description: item.MainDicomTags?.SeriesDescription ?? null,
      seriesNumber: item.MainDicomTags?.SeriesNumber ?? null,
      instances: Array.isArray(item.Instances) ? item.Instances : [],
    }));
  }

  async function getInstanceSearchRows(limit = 150) {
    const response = await findResources('Instance', {
      Query: {},
      Limit: limit,
      OrderBy: [
        {
          Type: 'DicomTag',
          Key: 'StudyDate',
          Direction: 'DESC',
        },
      ],
      RequestedTags: [
        'PatientID',
        'PatientName',
        'StudyID',
        'StudyDate',
        'StudyDescription',
        'StudyInstanceUID',
        'SeriesDescription',
        'SeriesInstanceUID',
        'SeriesNumber',
        'Modality',
        'InstanceNumber',
        'SliceThickness',
        'KVP',
        'XRayTubeCurrent',
        'TubeCurrent',
        'ReconstructionAlgorithm',
        'ExposureIndex',
        'TargetExposureIndex',
        'DeviationIndex',
      ],
    });

    return Array.isArray(response) ? response : [];
  }

  function buildStudyProfile(study, series, instance, instanceTags) {
    const studyTags = study?.MainDicomTags || {};
    const seriesTags = series?.MainDicomTags || {};
    const tags = instanceTags || {};

    return {
      id: study?.ID ?? null,
      patientId: normalizeValue(readTag(study, 'PatientID')),
      patientName: normalizeValue(readTag(study, 'PatientName')),
      studyId: normalizeValue(readTag(study, 'StudyID')),
      studyDate: normalizeValue(readTag(study, 'StudyDate')),
      studyDescription: normalizeValue(readTag(study, 'StudyDescription')),
      studyInstanceUID: normalizeValue(readTag(study, 'StudyInstanceUID')),
      modalitiesInStudy: normalizeValue(readTag(study, 'ModalitiesInStudy')),
      seriesCount: Array.isArray(study?.Series) ? study.Series.length : null,
      primarySeries: {
        id: series?.ID ?? null,
        seriesDate: normalizeValue(seriesTags.SeriesDate),
        seriesDescription: normalizeValue(seriesTags.SeriesDescription),
        seriesNumber: normalizeValue(seriesTags.SeriesNumber),
        modality: normalizeValue(seriesTags.Modality),
        protocolName: normalizeValue(seriesTags.ProtocolName),
      },
      primaryInstance: {
        id: instance?.ID ?? null,
        instanceNumber: normalizeValue(tags.InstanceNumber),
        sliceThickness: normalizeValue(tags.SliceThickness),
        kvp: normalizeValue(tags.KVP),
        tubeCurrent: normalizeValue(tags.XRayTubeCurrent ?? tags.TubeCurrent),
        reconstructionAlgorithm: normalizeValue(tags.ReconstructionAlgorithm),
        exposureIndex: normalizeValue(tags.ExposureIndex),
        targetExposureIndex: normalizeValue(tags.TargetExposureIndex),
        deviationIndex: normalizeValue(tags.DeviationIndex),
      },
      raw: {
        study: studyTags,
        series: seriesTags,
        instanceTags: tags,
      },
    };
  }

  async function getHealth() {
    try {
      const system = await getSystem();
      return {
        connected: true,
        status: 'Orthanc: Connected',
        name: system?.Name ?? system?.name ?? 'Orthanc',
        version: system?.Version ?? system?.version ?? null,
      };
    } catch {
      return {
        connected: false,
        status: 'Orthanc: Not Connected',
        name: null,
        version: null,
      };
    }
  }

  async function getDashboardSummary() {
    try {
      const health = await getHealth();
      if (!health.connected) {
        return {
          ...health,
          studies: null,
          series: null,
          instances: null,
          recentActivity: [],
          message: 'Unable to retrieve Orthanc data.',
        };
      }

      const [statistics, changes] = await Promise.all([getStatistics(), getChanges(6)]);
      return {
        ...health,
        studies: statistics.CountStudies ?? statistics.Studies ?? null,
        series: statistics.CountSeries ?? statistics.Series ?? null,
        instances: statistics.CountInstances ?? statistics.Instances ?? null,
        recentActivity: Array.isArray(changes?.Changes) ? changes.Changes.slice(0, 6).map(formatActivity) : [],
        message: null,
      };
    } catch {
      return {
        connected: false,
        status: 'Orthanc unavailable',
        name: null,
        version: null,
        studies: null,
        series: null,
        instances: null,
        recentActivity: [],
        message: 'Unable to retrieve Orthanc data.',
      };
    }
  }

  async function listStudies({ limit = 50 } = {}) {
    try {
      const rows = await findResources('Study', {
        Query: {},
        Limit: Math.min(Math.max(Number(limit) || 50, 1), 200),
        OrderBy: [
          {
            Type: 'DicomTag',
            Key: 'StudyDate',
            Direction: 'DESC',
          },
        ],
        RequestedTags: ['PatientID', 'PatientName', 'StudyID', 'StudyDate', 'StudyDescription', 'StudyInstanceUID', 'ModalitiesInStudy'],
      });

      return rows.map((study) => ({
        id: study.ID,
        patientId: normalizeValue(readTag(study, 'PatientID')),
        patientName: normalizeValue(readTag(study, 'PatientName')),
        studyId: normalizeValue(readTag(study, 'StudyID')),
        studyDate: normalizeValue(readTag(study, 'StudyDate')),
        studyDescription: normalizeValue(readTag(study, 'StudyDescription')),
        studyInstanceUID: normalizeValue(readTag(study, 'StudyInstanceUID')),
        modalitiesInStudy: normalizeValue(readTag(study, 'ModalitiesInStudy')),
        seriesCount: Array.isArray(study.Series) ? study.Series.length : null,
      }));
    } catch {
      try {
        const ids = await request('GET', '/studies');
        const limitedIds = Array.isArray(ids) ? ids.slice(0, Math.min(Math.max(Number(limit) || 50, 1), 200)) : [];
        const studies = await Promise.all(limitedIds.map((studyId) => getStudy(studyId).catch(() => null)));

        return studies.filter(Boolean).map((study) => ({
          id: study.ID,
          patientId: normalizeValue(readTag(study, 'PatientID')),
          patientName: normalizeValue(readTag(study, 'PatientName')),
          studyId: normalizeValue(readTag(study, 'StudyID')),
          studyDate: normalizeValue(readTag(study, 'StudyDate')),
          studyDescription: normalizeValue(readTag(study, 'StudyDescription')),
          studyInstanceUID: normalizeValue(readTag(study, 'StudyInstanceUID')),
          modalitiesInStudy: normalizeValue(readTag(study, 'ModalitiesInStudy')),
          seriesCount: Array.isArray(study.Series) ? study.Series.length : null,
        }));
      } catch {
        return [];
      }
    }
  }

  async function getExposureIndexRows({ limit = 150 } = {}) {
    try {
      const rows = await getInstanceSearchRows(limit);

      return rows.map((instance) => ({
        id: instance.ID,
        studyId: normalizeValue(readTag(instance, 'StudyID')),
        studyDate: normalizeValue(readTag(instance, 'StudyDate')),
        studyDescription: normalizeValue(readTag(instance, 'StudyDescription')),
        studyInstanceUID: normalizeValue(readTag(instance, 'StudyInstanceUID')),
        patientId: normalizeValue(readTag(instance, 'PatientID')),
        patientName: normalizeValue(readTag(instance, 'PatientName')),
        modality: normalizeValue(readTag(instance, 'Modality')),
        seriesId: instance.ParentSeries ?? null,
        seriesDescription: normalizeValue(readTag(instance, 'SeriesDescription')),
        seriesNumber: normalizeValue(readTag(instance, 'SeriesNumber')),
        instanceNumber: normalizeValue(readTag(instance, 'InstanceNumber')),
        exposureIndex: normalizeValue(readTag(instance, 'ExposureIndex')),
        targetExposureIndex: normalizeValue(readTag(instance, 'TargetExposureIndex')),
        deviationIndex: normalizeValue(readTag(instance, 'DeviationIndex')),
      }));
    } catch {
      return [];
    }
  }

  async function getStudyProfile(studyId) {
    const study = await getStudy(studyId);
    const seriesIds = Array.isArray(study.Series) ? study.Series : [];
    const primarySeriesId = seriesIds[0] ?? null;

    let series = null;
    let instance = null;
    let instanceTags = null;

    if (primarySeriesId) {
      series = await getSeries(primarySeriesId);
      const instanceIds = Array.isArray(series?.Instances) ? series.Instances : [];
      const primaryInstanceId = instanceIds[0] ?? null;

      if (primaryInstanceId) {
        instance = await getInstance(primaryInstanceId);
        instanceTags = await getInstanceTags(primaryInstanceId);
      }
    }

    return buildStudyProfile(study, series, instance, instanceTags);
  }

  async function compareStudies(leftStudyId, rightStudyId) {
    const [left, right] = await Promise.all([getStudyProfile(leftStudyId), getStudyProfile(rightStudyId)]);
    return { left, right };
  }

  async function getOrthancStatus() {
    const health = await getHealth();
    return health.connected
      ? { connected: true, status: health.status, name: health.name, version: health.version }
      : { connected: false, status: 'Orthanc unavailable', name: null, version: null };
  }

  async function getRecentActivity(limit = 6) {
    try {
      const changes = await getChanges(limit);
      return Array.isArray(changes?.Changes) ? changes.Changes.slice(0, limit).map(formatActivity) : [];

      return [];
    }
  }

  async function getConnectionSummary() {

    if (!health.connected) {

      studies: statistics.CountStudies ?? statistics.Studies ?? null,

      instances: statistics.CountInstances ?? null,
    };

    listSeriesForStudy,
    getInstancePreview,

    compareStudies,
