import jwt from 'jsonwebtoken';

export function createAuthMiddleware(jwtSecret, { getAuthState } = {}) {
  return function requireAuth(req, res, next) {
    const header = req.headers.authorization || '';
    const [scheme, token] = header.split(' ');

    if (scheme !== 'Bearer' || !token) {
      return res.status(401).json({ error: 'Authentication required.' });
    }

    let payload;
    try {
      payload = jwt.verify(token, jwtSecret);
    } catch {
      return res.status(401).json({ error: 'Session expired or invalid. Please sign in again.' });
    }

    req.user = {
      id: payload.sub,
      username: payload.username,
      role: payload.role,
    };

    // Re-check the account on every request so deactivation, restriction and role changes apply immediately.
    if (getAuthState) {
      const account = getAuthState(payload.sub);
      if (!account || account.status !== 'active' || account.restricted) {
        return res.status(401).json({ error: 'This account is no longer permitted to sign in.' });
      }
      if (Number(payload.tokenVersion ?? 0) !== Number(account.tokenVersion)) {
        return res.status(401).json({ error: 'Session expired or invalid. Please sign in again.' });
      }
      req.user = { id: account.id, username: account.username, role: account.role };
    }

    return next();
  };
}

export function createRoleMiddleware(allowedRoles) {
  const roles = Array.isArray(allowedRoles) ? allowedRoles : [allowedRoles];

  return function requireRole(req, res, next) {
    if (!req.user) {
      return res.status(401).json({ error: 'Authentication required.' });
    }

    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'You do not have permission to perform this action.' });
    }

    return next();
  };
}

import { Router } from 'express';
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { createLoginRateLimiter } from '../middleware/loginRateLimit.js';

function normalizeUsername(value) {
  return String(value || '').trim();
}

export function createAuthRouter({
  db,
  audit,
  alerts,
  users,
  jwtSecret,
  jwtExpiresIn,
  requireAuth,
  loginRateLimiter = createLoginRateLimiter(),
}) {
  const router = Router();
  const findUser = db.prepare(
    'SELECT id, username, password_hash, role, status, restricted, token_version FROM users WHERE username = ?',
  );
  const recentFailures = db.prepare(
    `SELECT COUNT(*) AS total FROM audit_logs
     WHERE action = 'login_failed' AND description = ? AND timestamp >= datetime('now', '-15 minutes')`,
  );

  router.post('/login', (req, res) => {
    const clientKey = loginRateLimiter.clientKey(req);
    const retryAfter = loginRateLimiter.retryAfter(clientKey);
    if (retryAfter) {
      res.setHeader('Retry-After', String(retryAfter));
      return res.status(429).json({ error: 'Too many sign-in attempts. Please try again later.' });
    }

    const username = normalizeUsername(req.body?.username);
    const password = req.body?.password;

    if (!username || typeof password !== 'string' || password.length === 0) {
      return res.status(400).json({ error: 'Username and password are required.' });
    }

    const user = findUser.get(username);
    const passwordValid = user ? bcrypt.compareSync(password, user.password_hash) : false;

    if (!user || !passwordValid) {
      loginRateLimiter.recordFailure(clientKey);
      const description = `Failed login attempt for username "${username}".`;
      audit.log({
        userId: user?.id ?? null,
        action: 'login_failed',
        target: 'auth',
        description,
      });
      if (alerts && recentFailures.get(description).total >= 3) {
        alerts.raise({
          type: 'access',
          description: `Repeated failed sign-in attempts for username "${username}" in the last 15 minutes.`,
          resource: `user:${username}`,
          dedupeMinutes: 15,
        });
      }
      return res.status(401).json({ error: 'Invalid username or password.' });
    }

    if (user.status !== 'active' || user.restricted) {
      audit.log({
        userId: user.id,
        action: 'login_blocked',
        target: 'auth',
        description: `Sign-in blocked for "${user.username}" (account ${user.restricted ? 'restricted' : user.status}).`,
      });
      return res.status(403).json({
        error: user.restricted
          ? 'This account is restricted. Contact an administrator.'
          : 'This account is inactive. Contact an administrator.',
      });
    }

    const token = jwt.sign(
      {
        sub: user.id,
        username: user.username,
        role: user.role,
        tokenVersion: user.token_version,
      },
      jwtSecret,
      { expiresIn: jwtExpiresIn },
    );

    audit.log({
      userId: user.id,
      action: 'LOGIN',
      target: 'auth',
      description: `User "${user.username}" signed in.`,
    });
    users?.recordLogin(user.id);
    loginRateLimiter.reset(clientKey);

    return res.json({
      token,
      user: {
        id: user.id,
        username: user.username,
        role: user.role,
      },
    });
  });

  router.post('/logout', requireAuth, (req, res) => {
    users.revokeTokens(req.user.id);
    audit.log({
      userId: req.user.id,
      action: 'LOGOUT',
      target: 'auth',
      description: `User "${req.user.username}" signed out.`,
    });

    return res.json({ ok: true });
  });

  router.get('/me', requireAuth, (req, res) => {
    return res.json({
      user: req.user,
    });
  });

  router.patch('/me', requireAuth, (req, res) => {
    const { full_name, email, phone, theme_preference, notification_preference } = req.body || {};
    
    const updates = [];
    const values = [];
    
    if (full_name !== undefined) {
      updates.push('full_name = ?');
      values.push(full_name);
    }
    if (email !== undefined) {
      updates.push('email = ?');
      values.push(email);
    }
    if (phone !== undefined) {
      updates.push('phone = ?');
      values.push(phone);
    }
    if (theme_preference !== undefined) {
      updates.push('theme_preference = ?');
      values.push(theme_preference);
    }
    if (notification_preference !== undefined) {
      updates.push('notification_preference = ?');
      values.push(notification_preference ? 1 : 0);
    }
    
    if (updates.length === 0) {
      return res.status(400).json({ error: 'No fields to update.' });
    }
    
    updates.push('updated_at = datetime("now")');
    values.push(req.user.id);
    
    const stmt = db.prepare(`UPDATE users SET ${updates.join(', ')} WHERE id = ?`);
    stmt.run(...values);
    
    audit.log({
      userId: req.user.id,
      action: 'profile_update',
      target: `user:${req.user.id}`,
      description: `User "${req.user.username}" updated their profile.`,
    });
    
    const updatedUser = db.prepare(
      'SELECT id, username, full_name, email, phone, role, theme_preference, notification_preference FROM users WHERE id = ?'
    ).get(req.user.id);
    
    return res.json({ user: updatedUser });
  });

  router.post('/change-password', requireAuth, async (req, res) => {
    const { currentPassword, newPassword } = req.body || {};
    
    if (!currentPassword || !newPassword) {
      return res.status(400).json({ error: 'Current password and new password are required.' });
    }
    
    if (newPassword.length < 8) {
      return res.status(400).json({ error: 'New password must be at least 8 characters long.' });
    }
    
    const user = db.prepare('SELECT id, password_hash, token_version FROM users WHERE id = ?').get(req.user.id);
    
    const passwordValid = bcrypt.compareSync(currentPassword, user.password_hash);
    if (!passwordValid) {
      return res.status(401).json({ error: 'Current password is incorrect.' });
    }
    
    const newPasswordHash = await bcrypt.hash(newPassword, 10);
    const newTokenVersion = (user.token_version || 0) + 1;
    
    db.prepare(
      'UPDATE users SET password_hash = ?, token_version = ?, updated_at = datetime("now") WHERE id = ?'
    ).run(newPasswordHash, newTokenVersion, req.user.id);
    
    audit.log({
      userId: req.user.id,
      action: 'password_change',
      target: `user:${req.user.id}`,
      description: `User "${req.user.username}" changed their password.`,
    });
    
    return res.json({ ok: true });
  });

  return router;
}

function withTemporaryDatabase(run) {
  const directory = fs.mkdtempSync(path.join(os.tmpdir(), 'orthoviewplus-test-'));
  const databasePath = path.join(directory, 'test.db');
  const db = createDatabase(databasePath);

  const cleanup = () => {
    db.close();
    fs.rmSync(directory, { recursive: true, force: true });
  };
  try {
    const result = run(db);
    if (result && typeof result.then === 'function') {
      return result.finally(cleanup);
    }
    cleanup();
    return result;
  } catch (error) {
    cleanup();
    throw error;
  }
}

async function withAuthApp(db, { loginRateLimiter } = {}, run) {
  const users = createUserService(db);
  const audit = createAuditService(db);
  const jwtSecret = 'test-secret-key-used-only-by-this-test';
  const requireAuth = createAuthMiddleware(jwtSecret, { getAuthState: users.getAuthState });
  const router = createAuthRouter({
    db,
    audit,
    alerts: createAlertService(db),
    users,
    jwtSecret,
    jwtExpiresIn: '8h',
    requireAuth,
    loginRateLimiter,
  });
  const app = express();
  app.use(express.json());
  app.use('/auth', router);
  app.get('/protected', requireAuth, (req, res) => res.json({ userId: req.user.id }));
  const server = app.listen(0, '127.0.0.1');
  await new Promise((resolve) => server.once('listening', resolve));
  try {
    await run(`http://127.0.0.1:${server.address().port}`);
  } finally {
    server.closeAllConnections();
    await new Promise((resolve, reject) => server.close((error) => error ? reject(error) : resolve()));
  }
}

      assert.equal(afterSuccess.status, 401);
    });
  });

function withTemporaryDatabase(run) {
  const directory = fs.mkdtempSync(path.join(os.tmpdir(), 'orthoviewplus-test-'));
  const databasePath = path.join(directory, 'test.db');
  const db = createDatabase(databasePath);

  try {
    run(db);
  } finally {
    db.close();
    fs.rmSync(directory, { recursive: true, force: true });
  }
}

test('authorization: role middleware allows administrator role', () => {
  withTemporaryDatabase((db) => {
    const users = createUserService(db);
    const requireAdministrator = createRoleMiddleware('administrator');

    const user = users.create({
      username: 'admin',
      password: 'TestPassword123',
      role: 'administrator',
    });

    const req = { user: { id: user.id, username: 'admin', role: 'administrator' } };
    const res = {
      status: (code) => ({ json: (data) => ({ status: code, data }) }),
    };
    let nextCalled = false;
    const next = () => { nextCalled = true; };

    requireAdministrator(req, res, next);

    assert.equal(nextCalled, true);
  });
});

test('authorization: role middleware accepts multiple allowed roles', () => {
  withTemporaryDatabase((db) => {
    const users = createUserService(db);
    const requireAdminOrPractitioner = createRoleMiddleware(['administrator', 'practitioner']);

    const admin = users.create({
      username: 'admin',
      password: 'TestPassword123',
      role: 'administrator',
    });

    const practitioner = users.create({
      username: 'practitioner',
      password: 'TestPassword123',
      role: 'practitioner',
    });

    // Test administrator
    const req1 = { user: { id: admin.id, username: 'admin', role: 'administrator' } };
    const res1 = {
      status: (code) => ({ json: (data) => ({ status: code, data }) }),
    };
    let nextCalled1 = false;
    const next1 = () => { nextCalled1 = true; };

    requireAdminOrPractitioner(req1, res1, next1);
    assert.equal(nextCalled1, true);

    // Test practitioner
    const req2 = { user: { id: practitioner.id, username: 'practitioner', role: 'practitioner' } };
    const res2 = {
      status: (code) => ({ json: (data) => ({ status: code, data }) }),
    };
    let nextCalled2 = false;
    const next2 = () => { nextCalled2 = true; };

    requireAdminOrPractitioner(req2, res2, next2);
    assert.equal(nextCalled2, true);
  });
});
